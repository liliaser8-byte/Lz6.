*Програмний код:*

```cpp
	private: System::Void button1_Click(System::Object^ sender, System::EventArgs^ e) {
		// 1. Оголошення змінних та масиву для збереження введених даних
		double arr[5];

		// 2. Перевірка введених даних (валідація)
		if (!System::Double::TryParse(textBox1->Text, arr[0]) ||
			!System::Double::TryParse(textBox2->Text, arr[1]) ||
			!System::Double::TryParse(textBox3->Text, arr[2]) ||
			!System::Double::TryParse(textBox4->Text, arr[3]) ||
			!System::Double::TryParse(textBox5->Text, arr[4]))
		{
			System::Windows::Forms::MessageBox::Show(
				"Будь ласка, введіть коректні числові значення в усі поля (у см)!",
				"Помилка введення"
			);
			return; // Вихід з методу в разі помилки (дозволяє виправити дані та спробувати знову)
		}

		// 3. Створення та використання щонайменше двох звичайних вказівників C++
		double* ptr = arr;              // Перший вказівник: на початок масиву
		double* start_ptr = &arr[0];     // Другий вказівник: на перше (початкове) вимірювання
		double* end_ptr = &arr[4];       // Третій вказівник: на останнє (кінцеве) вимірювання

		// 4. Отримання початкового та кінцевого рівнів (розіменуванням вказівників)
		double initial_level = *start_ptr;
		double final_level = *end_ptr;

		// 5. Пошук максимального та мінімального рівнів
		double max_level = *ptr;
		double min_level = *ptr;
		for (int i = 1; i < 5; i++) {
			double current = *(ptr + i); // Розіменування вказівника зі зсувом
			if (current > max_level) {
				max_level = current;
			}
			if (current < min_level) {
				min_level = current;
			}
		}

		// 6. Обчислення середнього рівня
		double sum = 0;
		for (int i = 0; i < 5; i++) {
			sum += *(ptr + i); // Накопичення суми через розіменування
		}
		double average_level = sum / 5.0;

		// 7. Визначення кількості збільшень рівня порівняно з попереднім
		int increase_count = 0;
		for (int i = 1; i < 5; i++) {
			// Порівняння поточного значення з попереднім через арифметику вказівників
			if (*(ptr + i) > *(ptr + i - 1)) {
				increase_count++;
			}
		}

		// 8. Пошук найбільшої зміни за абсолютною величиною між сусідніми вимірюваннями
		double max_change = 0;
		for (int i = 1; i < 5; i++) {
			// Обчислення різниці між сусідніми елементами
			double diff = *(ptr + i) - *(ptr + i - 1);
			if (diff < 0) {
				diff = -diff; // Отримання абсолютної величини (модуля)
			}

			if (diff > max_change) {
				max_change = diff;
			}
		}

		// 9. Виведення результатів та пояснення розрахунків у Label
		label6->Text =
			"Результати аналізу:\n" +
			"• Початковий рівень: " + initial_level.ToString("F1") + " см, " +
			"Кінцевий рівень: " + final_level.ToString("F1") + " см\n" +
			"• Максимальний рівень: " + max_level.ToString("F1") + " см, " +
			"Мінімальний рівень: " + min_level.ToString("F1") + " см\n" +
			"• Середній рівень: " + average_level.ToString("F1") + " см\n" +
			"• Рівень води збільшувався порівняно з попереднім: " + increase_count.ToString() + " раз(и)\n" +
			"• Найбільша зміна між сусідніми вимірюваннями: " + max_change.ToString("F1") + " см";
	}
};
}

```

**Коментарі до виконання вимог:**
1. Windows Forms C++/CLI: Програма базується на формі з полями введення та кнопкою.
2. Перевірка даних: Використано конструкцію TryParse для перевірки наявності помилок у текстових полях перед розрахунками.
3. Вказівники C++: Створено вказівники типу double* (ptr, start_ptr, end_ptr).
4. Розіменування: Усі операції зчитування з масиву (пошук мінімуму, максимуму, суми та порівняння) відбуваються виключно через синтаксис розіменування *(ptr + i) замість традиційного індексного доступу arr[i].
5. Повторне виконання: Користувач може в будь-який момент змінити цифри в текстових полях та знову натиснути «Проаналізувати» — програма миттєво оновить розрахунки без необхідності перезавантаження.

**Короткий опис розробленої програми**
Програма «Аналіз рівня води» — це десктопний додаток (Windows Forms C++/CLI), призначений для обробки та аналізу даних про п'ять послідовних вимірювань рівня води в резервуарі. Основний функціонал програми:

* Зчитування 5 числових значень (у сантиметрах) із текстових полів після натискання кнопки.

* Збереження введених даних у масив, подальше опрацювання якого здійснюється виключно через застосування щонайменше двох звичайних вказівників C++.

* Виконання розрахунків шляхом розіменування вказівників для визначення: початкового та кінцевого рівнів, максимального і мінімального значень, середнього рівня, кількості зростань рівня води та найбільшої абсолютної зміни між сусідніми замірами.

* Виведення отриманих результатів у відповідне текстове поле на формі з коротким поясненням розрахунків.

* Забезпечення можливості багаторазового введення нових даних і повторного виконання обчислень без необхідності перезапуску самої програми.
